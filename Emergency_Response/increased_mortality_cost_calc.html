Reminder: \(Delta = Flood - Baseline\)

  • positive Delta = Baseline is faster

  • negative Delta = Detour (no bridge) is faster

# round delta time value to the nearest whole number
min_comp <- minute_comparison_original |>
  filter(metric == "Travel time (minutes)") |>
  mutate(baseline_time = round(baseline_value, digits = 0),
         flood_time = round(alternative_value, digits = 0),
         delta_round = round(delta, digits = 0)) |>
  group_by(block_geoid_2020) |>
  slice_min(alternative_value, n = 1, with_ties = FALSE) %>%
  ungroup() |>
  filter(delta >= 0)

min_comp_rr <- min_comp %>%
  left_join(rr,by = c("baseline_time" = "response_time_min")) %>%
  rename(rr_baseline = relative_risk_ratio) %>%
  left_join(rr,by = c("flood_time" = "response_time_min")) %>%
  rename(rr_flood = relative_risk_ratio) %>% 
  mutate(rr_ratio = rr_flood/rr_baseline)
